<script>
const pieces = ["♙","♗","♘","♖","♔","♕"];
let isRolling = false;

// reel 초기 생성
function createReel(reel) {
  reel.innerHTML = "";
  for (let i = 0; i < 30; i++) {
    const div = document.createElement("div");
    div.className = "symbol";
    div.textContent = pieces[Math.floor(Math.random()*6)];
    reel.appendChild(div);
  }
}

// 모든 릴 초기화
const reels = [
  document.getElementById("reel0"),
  document.getElementById("reel1"),
  document.getElementById("reel2")
];

reels.forEach(createReel);

function spinReel(reel, duration) {
  return new Promise(resolve => {
    let position = 0;
    const speed = 25; // 속도

    const interval = setInterval(() => {
      position += speed;
      reel.style.transform = `translateY(${position}px)`;

      // 끝까지 가면 반복 (무한 스크롤 느낌)
      if (position >= reel.scrollHeight / 2) {
        position = 0;
      }
    }, 16);

    setTimeout(() => {
      clearInterval(interval);

      // 최종 결과 정렬
      const finalIndex = Math.floor(Math.random() * 6);
      const finalSymbol = pieces[finalIndex];

      reel.innerHTML = "";
      for (let i = 0; i < 10; i++) {
        const div = document.createElement("div");
        div.className = "symbol";
        div.textContent = i === 5 ? finalSymbol : pieces[Math.floor(Math.random()*6)];
        reel.appendChild(div);
      }

      reel.style.transform = "translateY(-600px)";
      resolve();
    }, duration);
  });
}

async function roll() {
  if (isRolling) return;
  isRolling = true;

  reels.forEach(createReel);

  await Promise.all([
    spinReel(reels[0], 1500),
    spinReel(reels[1], 2000),
    spinReel(reels[2], 2500)
  ]);

  isRolling = false;
}

// 터치 anywhere
document.body.addEventListener("click", roll);
</script>
